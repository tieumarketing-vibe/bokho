function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    var doc = SpreadsheetApp.getActiveSpreadsheet();
    
    // LOG để debug (có thể xem trong Apps Script execution log)
    Logger.log("Received POST request");
    Logger.log("Content: " + e.postData.contents);
    
    // Lấy dữ liệu (xử lý cả JSON lẫn form-urlencoded)
    var data;
    if (e.postData.type === 'application/x-www-form-urlencoded') {
      data = e.parameter;
    } else {
      try {
        data = JSON.parse(e.postData.contents);
      } catch (parseError) {
        data = e.parameter;
      }
    }
    
    Logger.log("Parsed data: " + JSON.stringify(data));
    
    // XÁC ĐỊNH TÊN SHEET
    var sheetName = "orderform";
    if (data.formType === 'contact') {
      sheetName = "Form contact";
    }
    
    Logger.log("Target sheet: " + sheetName);

    var sheet = doc.getSheetByName(sheetName);
    if (!sheet) {
      Logger.log("Sheet not found, creating new sheet");
      sheet = doc.insertSheet(sheetName);
    }

    // GHI DỮ LIỆU
    if (data.formType === 'contact') {
      if (sheet.getLastRow() === 0) {
        var headerRow = ["Thời gian", "Họ tên", "Số điện thoại", "Email", "Nội dung"];
        sheet.getRange(1, 1, 1, 5).setValues([headerRow]);
        sheet.getRange(1, 1, 1, 5).setFontWeight("bold");
      }
      sheet.appendRow([new Date(), data.name || '', "'" + (data.phone || ''), data.email || '', data.message || '']);
      Logger.log("Contact form data appended");
      
    } else {
      if (sheet.getLastRow() === 0) {
        var headerRow = ["Thời gian", "Họ tên", "Số điện thoại", "Địa chỉ", "Sản phẩm", "Số lượng", "Ghi chú", "Đơn giá", "Thành tiền"];
        sheet.getRange(1, 1, 1, 9).setValues([headerRow]);
        sheet.getRange(1, 1, 1, 9).setFontWeight("bold");
      }
      
      sheet.appendRow([
        new Date(), 
        data.name || '', 
        "'" + (data.phone || ''), 
        data.address || '', 
        data.product || '', 
        data.quantity || '', 
        data.note || '',
        data.price || '',
        data.total || ''
      ]);
      Logger.log("Order form data appended");
    }

    return ContentService
      .createTextOutput(JSON.stringify({ "result": "success", "sheet": sheetName }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    Logger.log("Error occurred: " + error.toString());
    return ContentService
      .createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  return ContentService.createTextOutput("Web App is running. POST requests are being accepted.");
}
