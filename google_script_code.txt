// HƯỚNG DẪN CẬP NHẬT GOOGLE APPS SCRIPT (CHIA SHEET THEO YÊU CẦU)
// =========================================================

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    var doc = SpreadsheetApp.getActiveSpreadsheet();
    
    // Lấy dữ liệu
    var data;
    try {
      data = JSON.parse(e.postData.contents);
    } catch (parseError) {
      data = e.parameter;
    }
    
    // XÁC ĐỊNH TÊN SHEET DỰA VÀO LOẠI FORM
    var sheetName = "orderform"; // Mặc định là sheet đơn hàng
    
    if (data.formType === 'contact') {
      sheetName = "Form contact";
    } else if (data.formType === 'order') {
      sheetName = "orderform";
    }

    var sheet = doc.getSheetByName(sheetName);
    // Nếu chưa có sheet thì tự tạo
    if (!sheet) {
      sheet = doc.insertSheet(sheetName);
    }

    // GHI DỮ LIỆU
    if (data.formType === 'contact') {
      // --- FORM LIÊN HỆ (Sheet: Form contact) ---
      if (sheet.getLastRow() === 0) {
        var headerRow = ["Thời gian", "Họ tên", "Số điện thoại", "Email", "Nội dung"];
        sheet.getRange(1, 1, 1, 5).setValues([headerRow]);
        sheet.getRange(1, 1, 1, 5).setFontWeight("bold");
      }
      sheet.appendRow([new Date(), data.name, "'" + data.phone, data.email, data.message]);
      
    } else {
      // --- FORM ĐẶT HÀNG (Sheet: orderform) ---
      if (sheet.getLastRow() === 0) {
        var headerRow = ["Thời gian", "Họ tên", "Số điện thoại", "Địa chỉ", "Sản phẩm", "Số lượng", "Ghi chú", "Đơn giá", "Thành tiền"];
        sheet.getRange(1, 1, 1, 9).setValues([headerRow]);
        sheet.getRange(1, 1, 1, 9).setFontWeight("bold");
      }
      
      sheet.appendRow([
        new Date(), 
        data.name, 
        "'" + data.phone, 
        data.address, 
        data.product, 
        data.quantity, 
        data.note,
        data.price,
        data.total
      ]);
    }

    return ContentService.createTextOutput(JSON.stringify({ "result": "success" })).setMimeType(ContentService.MimeType.JSON);

  } catch (e) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": e.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
