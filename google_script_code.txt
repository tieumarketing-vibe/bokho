// =========================================================

// Vấn đề: Google chặn các trang web lạ (như GitHub) nếu bạn không mở quyền "Bất kỳ ai".
// Giải pháp: Cập nhật code này và Cấu hình lại quyền truy cập.

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    var doc = SpreadsheetApp.getActiveSpreadsheet();
    
    // Lấy dữ liệu: Xử lý trường hợp gửi từ fetch với text/plain
    var data;
    try {
      data = JSON.parse(e.postData.contents);
    } catch (parseError) {
      // Fallback nếu dữ liệu gửi dạng form-urlencoded (ít gặp với code hiện tại nhưng an toàn hơn)
      data = e.parameter;
    }
    
    // XÁC ĐỊNH LOẠI FORM
    var sheetName = "Sheet1"; 
    if (data.formType === 'contact') {
      sheetName = "Form contact";
    }

    var sheet = doc.getSheetByName(sheetName);
    if (!sheet) {
      sheet = doc.insertSheet(sheetName);
    }

    // GHI DỮ LIỆU
    if (data.formType === 'contact') {
      if (sheet.getLastRow() === 0) {
        var headerRow = ["Thời gian", "Họ tên", "Số điện thoại", "Email", "Nội dung"];
        sheet.getRange(1, 1, 1, 5).setValues([headerRow]);
        sheet.getRange(1, 1, 1, 5).setFontWeight("bold");
      }
      sheet.appendRow([new Date(), data.name, "'" + data.phone, data.email, data.message]);
    } else {
      if (sheet.getLastRow() === 0) {
        var headerRow = ["Thời gian", "Họ tên", "Số điện thoại", "Địa chỉ", "Sản phẩm", "Số lượng", "Ghi chú"];
        sheet.getRange(1, 1, 1, 7).setValues([headerRow]);
        sheet.getRange(1, 1, 1, 7).setFontWeight("bold");
      }
      sheet.appendRow([new Date(), data.name, "'" + data.phone, data.address, data.product, data.quantity, data.note]);
    }

    return ContentService.createTextOutput(JSON.stringify({ "result": "success" })).setMimeType(ContentService.MimeType.JSON);

  } catch (e) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": e.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

// Hàm này để kiểm tra xem Link có hoạt động không
function doGet(e) {
  return ContentService.createTextOutput("Kết nối thành công! Web App đã được cấu hình đúng quyền truy cập.");
}
